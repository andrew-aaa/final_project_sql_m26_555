## Прошлое задание и создание таблиц

```bash
DROP TABLE IF EXISTS PRODUCT CASCADE;
DROP TABLE IF EXISTS RACK CASCADE;
DROP TABLE IF EXISTS CLIENTS CASCADE;
DROP TABLE IF EXISTS PREMISES CASCADE;

CREATE TABLE PREMISES (
	ID_PREMISE INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
	NAME VARCHAR(100) NOT NULL UNIQUE,
	USABLE_VOLUME NUMERIC(12, 2) NOT NULL CHECK (USABLE_VOLUME > 0),
	MIN_TEMPERATURE NUMERIC(5, 2) NOT NULL,
	MAX_TEMPERATURE NUMERIC(5, 2) NOT NULL,
	MIN_HUMIDITY NUMERIC(5, 2) NOT NULL,
	MAX_HUMIDITY NUMERIC(5, 2) NOT NULL,
	CONSTRAINT CHK_PREMISES_TEMP CHECK (
		MIN_TEMPERATURE >= -100.00 
		AND MAX_TEMPERATURE <= 100.00 
		AND MIN_TEMPERATURE <= MAX_TEMPERATURE
	),
	CONSTRAINT CHK_PREMISES_HUMIDITY CHECK (
		MIN_HUMIDITY >= 0.00 
		AND MAX_HUMIDITY <= 100.00 
		AND MIN_HUMIDITY <= MAX_HUMIDITY
	)
);

CREATE TABLE CLIENTS (
	ID_CLIENT INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
	NAME_LEGAL_ENTITY VARCHAR(200) NOT NULL,
	BANK_DETAILS TEXT NOT NULL
);

CREATE TABLE RACK (
	ID_RACK INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
	RACK_NUMBER VARCHAR(20) NOT NULL,
	ID_PREMISE INTEGER NOT NULL REFERENCES PREMISES (ID_PREMISE),
	NUMBER_OF_PLACES INTEGER NOT NULL CHECK (NUMBER_OF_PLACES > 0),
	HEIGHT NUMERIC(8, 2) NOT NULL CHECK (HEIGHT > 0),
	WIDTH NUMERIC(8, 2) NOT NULL CHECK (WIDTH > 0),
	LENGTHS NUMERIC(8, 2) NOT NULL CHECK (LENGTHS > 0),
	MAX_LOAD NUMERIC(8, 2) NOT NULL CHECK (MAX_LOAD > 0),
	CONSTRAINT UQ_RACK_NUMBER_PER_PREMISES UNIQUE (ID_PREMISE, RACK_NUMBER)
);

CREATE TABLE PRODUCT (
	ID_PRODUCT INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
	ID_RACK INTEGER NOT NULL REFERENCES RACK (ID_RACK),
	ID_CLIENT INTEGER NOT NULL REFERENCES CLIENTS (ID_CLIENT),
	HEIGHT NUMERIC(10, 3) NOT NULL CHECK (HEIGHT > 0),
	WIDTH NUMERIC(10, 3) NOT NULL CHECK (WIDTH > 0),
	LENGTHS NUMERIC(10, 3) NOT NULL CHECK (LENGTHS > 0),
	WEIGHT NUMERIC(12, 3) NOT NULL CHECK (WEIGHT > 0),
	DATE_OF_RECEIPT DATE NOT NULL DEFAULT CURRENT_DATE,
	CONTRACT_NUMBER VARCHAR(50) NOT NULL,
	END_DATE DATE NOT NULL,
	MIN_TEMPERATURE NUMERIC(5, 2) NOT NULL,
	MAX_TEMPERATURE NUMERIC(5, 2) NOT NULL,
	MIN_HUMIDITY NUMERIC(5, 2) NOT NULL,
	MAX_HUMIDITY NUMERIC(5, 2) NOT NULL,
	POSITIONS INTEGER NOT NULL CHECK (POSITIONS > 0),
	CONSTRAINT CHK_PRODUCT_DATES CHECK (END_DATE >= DATE_OF_RECEIPT),
	CONSTRAINT CHK_PRODUCT_TEMP CHECK (
		MIN_TEMPERATURE >= -100.00 
		AND MAX_TEMPERATURE <= 100.00 
		AND MIN_TEMPERATURE <= MAX_TEMPERATURE
	),
	CONSTRAINT CHK_PRODUCT_HUMIDITY CHECK (
		MIN_HUMIDITY >= 0.00 
		AND MAX_HUMIDITY <= 100.00 
		AND MIN_HUMIDITY <= MAX_HUMIDITY
	),
	CONSTRAINT UQ_RACK_POSITION UNIQUE (ID_RACK, POSITIONS)
);
```
